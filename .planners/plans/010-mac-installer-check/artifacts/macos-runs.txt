  ^[[2mbash    ^[[0m~/.bash_completion.d/stanza
  ^[[0;36mhint    ^[[0madd to ~/.bash_profile: for f in ~/.bash_completion.d/*; do . "$f"; done
  ^[[2mzsh     ^[[0m~/.zsh/completions/_stanza

  ^[[0;33mwarn    ^[[0m~/.local/bin is not on your PATH
